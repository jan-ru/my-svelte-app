```ts
import { describe, expect, it } from 'vitest';
import { fit, squarify } from './size-treemap';

const area = (r: { w: number; h: number }) => r.w * r.h;
const items = [
	{ id: 'a', label: 'A', value: 6 },
	{ id: 'b', label: 'B', value: 6 },
	{ id: 'c', label: 'C', value: 4 },
	{ id: 'd', label: 'D', value: 3 },
	{ id: 'e', label: 'E', value: 2 },
	{ id: 'f', label: 'F', value: 2 },
	{ id: 'g', label: 'G', value: 1 }
];

describe('squarify', () => {
	it('fills the rectangle and gives each item an area proportional to its value', () => {
		const rects = squarify(items, 600, 400);
		expect(rects.reduce((s, r) => s + area(r), 0)).toBeCloseTo(600 * 400, 6);
		const total = items.reduce((s, i) => s + i.value, 0);
		for (const r of rects) {
			const v = items.find((i) => i.id === r.id)!.value;
			expect(area(r)).toBeCloseTo((600 * 400 * v) / total, 6);
		}
	});
	it('keeps every rectangle inside the bounds and never overlaps two', () => {
		const rects = squarify(items, 600, 400);
		for (const r of rects) {
			expect(r.x).toBeGreaterThanOrEqual(-1e-9);
			expect(r.y).toBeGreaterThanOrEqual(-1e-9);
			expect(r.x + r.w).toBeLessThanOrEqual(600 + 1e-9);
			expect(r.y + r.h).toBeLessThanOrEqual(400 + 1e-9);
		}
		for (const a of rects) {
			for (const b of rects) {
				if (a === b) continue;
				const ox = Math.min(a.x + a.w, b.x + b.w) - Math.max(a.x, b.x);
				const oy = Math.min(a.y + a.h, b.y + b.h) - Math.max(a.y, b.y);
				expect(ox > 1e-6 && oy > 1e-6).toBe(false);
			}
		}
	});
	it('puts the largest item first', () => {
		expect(squarify(items, 600, 400)[0].id).toMatch(/^[ab]$/);
	});
	it('handles one item, zero and negative values, and an empty input', () => {
```

```ts
export interface WbCountry {
	iso3: string;
	name: string;
}

/**
 * NYU country names that differ from the World Bank's. Checked against the live API on 2026-10-28; the build script
 * prints any target that no longer exists. Jersey and Guernsey are left out on purpose: the World Bank reports them
 * together as "Channel Islands", and mapping both would count that GDP twice. Taiwan and Martinique have no World Bank
 * entry. All of these stay in the company counts and show GDP as "n/a".
 */
export const MANUAL_NAMES: Record<string, string> = {
	'South Korea': 'Korea, Rep.',
	'Hong Kong': 'Hong Kong SAR, China',
	Turkey: 'Turkiye',
	Vietnam: 'Viet Nam',
	'Czech Republic': 'Czechia',
	'Ivory Coast': "Cote d'Ivoire",
	Egypt: 'Egypt, Arab Rep.',
	Venezuela: 'Venezuela, RB',
	'Palestinian Authority': 'West Bank and Gaza',
	'Trinidad & Tobago': 'Trinidad and Tobago',
	'Saint Lucia': 'St. Lucia',
	'Curaçao': 'Curacao',
	Bahamas: 'Bahamas, The',
	Macau: 'Macao SAR, China',
	Slovakia: 'Slovak Republic',
	Macedonia: 'North Macedonia',
	Russia: 'Russian Federation',
	'Turks & Caicos Islands': 'Turks and Caicos Islands'
};

export interface Matching {
	/** NYU country name to World Bank ISO3 code. */
	mapping: Record<string, string>;
	/** NYU names with no World Bank counterpart. */
	unmatched: string[];
	/** World Bank ISO3 codes no NYU name points at. */
	unusedWb: string[];
	/** NYU names whose manual target is not a World Bank name (the table needs fixing). */
	badManual: string[];
}

export function matchCountries(nyu: readonly string[], wb: readonly WbCountry[], manual: Record<string, string>): Matching {
	const byName = new Map(wb.map((c) => [c.name, c.iso3]));
	const mapping: Record<string, string> = {};
	const unmatched: string[] = [];
	const badManual: string[] = [];
	const owner = new Map<string, string>();
	for (const name of nyu) {
		let iso = byName.get(name);
		if (iso === undefined && name in manual) {
			iso = byName.get(manual[name]);
			if (iso === undefined) badManual.push(name);
		}
		if (iso === undefined) {
			unmatched.push(name);
			continue;
		}
		const other = owner.get(iso);
		if (other !== undefined) throw new Error(`${other} and ${name} both map to ${iso}`);
		owner.set(iso, name);
		mapping[name] = iso;
	}
	const used = new Set(Object.values(mapping));
	const unusedWb = wb.map((c) => c.iso3).filter((iso) => !used.has(iso)).sort();
	return { mapping, unmatched, unusedWb, badManual };
}
```

```ts
	source: string;

	countries: Record<string, { name: string; value: number; year: number }>;

	unmatched: string[];
}

/** GDP in current US$ and its year for an NYU country; null when unmapped or the World Bank has no value. */
export function gdpOf(data: GdpData, nyuName: string): { value: number; year: number } | null {
	const iso = data.nyu[nyuName];
	const country = iso ? data.countries[iso] : undefined;
	return country ? { value: country.value, year: country.year } : null;
}

/** NYU country name to GDP, for the countries that have one. */
export function gdpValues(data: GdpData, nyuNames: readonly string[]): Record<string, number> {
	const out: Record<string, number> = {};
	for (const name of nyuNames) {
		const gdp = gdpOf(data, name);
		if (gdp) out[name] = gdp.value;
	}
	return out;
}

/** Oldest and newest year among the countries' latest values. */
export function gdpYearRange(data: GdpData): [number, number] {
	const years = Object.values(data.countries).map((c) => c.year);
	return [Math.min(...years), Math.max(...years)];
}
```

```ts
	id: string;
	x: number;
	y: number;
	w: number;
	h: number;
}

/** Squarified treemap (Bruls, Huizing, van Wijk): rectangles with areas proportional to the values, as square as it can. Zero and negative values are dropped. */
export function squarify(items: readonly { id: string; value: number }[], width: number, height: number): Rect[] {
	const sorted = items.filter((i) => Number.isFinite(i.value) && i.value > 0).sort((a, b) => b.value - a.value);
	if (sorted.length === 0 || width <= 0 || height <= 0) return [];
	const total = sorted.reduce((s, i) => s + i.value, 0);
	const scale = (width * height) / total;
	const queue = sorted.map((i) => ({ id: i.id, area: i.value * scale }));
	const out: Rect[] = [];
	let x = 0;
	let y = 0;
	let w = width;
	let h = height;
	let row: { id: string; area: number }[] = [];

	const sum = (r: { area: number }[]) => r.reduce((s, n) => s + n.area, 0);
	const worst = (r: { area: number }[], side: number) => {
		const s = sum(r);
		const areas = r.map((n) => n.area);
		return Math.max((side * side * Math.max(...areas)) / (s * s), (s * s) / (side * side * Math.min(...areas)));
	};
	const layout = (r: { id: string; area: number }[]) => {
		const s = sum(r);
		if (w >= h) {
			// a column on the left, as tall as what is left
			const cw = s / h;
			let cy = y;
			for (const n of r) {
				out.push({ id: n.id, x, y: cy, w: cw, h: n.area / cw });
				cy += n.area / cw;
			}
			x += cw;
			w -= cw;
		} else {
			// a row on top, as wide as what is left
			const rh = s / w;
			let cx = x;
			for (const n of r) {
				out.push({ id: n.id, x: cx, y, w: n.area / rh, h: rh });
				cx += n.area / rh;
			}
			y += rh;
			h -= rh;
		}
	};

	while (queue.length > 0) {
		const side = Math.min(w, h);
		const next = queue[0];
		if (row.length === 0 || worst([...row, next], side) <= worst(row, side)) {
			row.push(queue.shift()!);
		} else {
			layout(row);
			row = [];
		}
	}
	if (row.length > 0) layout(row);
	return out;
}

/** A label shortened to fit a tile of the given width (about 6.5 px per character); '' when there is no room. */
export function fit(label: string, width: number, charWidth = 6.5): string {
	const max = Math.floor((width - 8) / charWidth);
	if (max < 3) return '';
	return label.length <= max ? label : `${label.slice(0, max - 1)}…`;
}
```

```ts
import { MEASURES, type Measure } from './size';

/** Every query parameter the size pages read; each must be in ISR_ALLOW_QUERY (a test checks). */
export const SIZE_PARAMS = ['measure', 'country', 'industry', 'sector'] as const;
export const MAX_SELECTED = 12;

export type SizePage = '/countries' | '/industries';
export interface SizeNav {
	measure: Measure;
	countries: string[];
	industries: string[];
	sectors: string[];
}
export interface SizeKnown {
	countries: readonly string[];
	industries: readonly string[];
	sectors: readonly string[];
}

const clean = (values: string[], known: readonly string[]): string[] =>
	[...new Set(values)]
		.filter((v) => known.includes(v))
		.sort((a, b) => a.localeCompare(b))
		.slice(0, MAX_SELECTED);

export function parseSizeNav(params: URLSearchParams, known: SizeKnown, page: SizePage): SizeNav {
	const countries = clean(params.getAll('country'), known.countries);
	const industries = clean(params.getAll('industry'), known.industries);
	const sectors = clean(params.getAll('sector'), known.sectors);
	const asked = MEASURES.find((m) => m.id === params.get('measure'))?.id ?? 'companies';
	// GDP exists only for countries, so it needs the countries page and no industry or sector restriction
	const measure: Measure = asked === 'gdp' && page === '/countries' && industries.length === 0 && sectors.length === 0 ? 'gdp' : 'companies';
	return { measure, countries, industries, sectors };
}

/** One canonical URL per view: the facets are sorted, so the order they were chosen in does not matter. */
export function sizeHref(page: SizePage, nav: SizeNav): string {
	const sp = new URLSearchParams();
	if (nav.measure !== 'companies') sp.set('measure', nav.measure);
	for (const c of nav.countries) sp.append('country', c);
	for (const i of nav.industries) sp.append('industry', i);
	for (const s of nav.sectors) sp.append('sector', s);
	const query = sp.toString();
	return query ? `${page}?${query}` : page;
}

export const toggle = (list: string[], value: string): string[] => (list.includes(value) ? list.filter((v) => v !== value) : [...list, value]);
```

```ts
import type { CompanyData } from './companies';
import { gdpValues, gdpYearRange, type GdpData } from './gdp';
import {
	COUNTRY_FILL,
	crossCell,
	countBy,
	dominantSector,
	sectorFill,
	shareOfGlobal,
	topIds,
	type CrossCell,
	type Shares
} from './size';
import type { SizeNav } from './size-nav';
import type { TreemapItem } from './size-treemap';

/** How many items the bars show when nothing is selected. */
export const TOP_N = 10;

export interface CountriesView {
	measureLabel: string;
	/** The global total the percentages refer to: all companies, or the World Bank world GDP. */
	total: number;
	/** Every country with a value for the measure. */
	tiles: TreemapItem[];
	/** The selected countries (or the largest ten) against the global total. */
	shares: Shares;
	/** The industries of the selected countries, by company count; null when no country is selected. */
	breakdown: TreemapItem[] | null;
	/** The global company count, the denominator for `breakdown`. */
	breakdownTotal: number;
	/** One country and one industry together. */
	cross: CrossCell | null;
	/** What the GDP numbers are; null for the company measure. */
	gdpNote: string | null;
}

const tilesOf = (values: Record<string, number>, fill: (id: string) => string): TreemapItem[] =>
	Object.entries(values)
		.filter(([, v]) => v > 0)
		.map(([id, value]) => ({ id, label: id, value, fill: fill(id) }))
		.sort((a, b) => b.value - a.value || a.label.localeCompare(b.label));

export function countriesView(data: CompanyData, gdp: GdpData, nav: SizeNav): CountriesView {
	const gdpMeasure = nav.measure === 'gdp';
	const values = gdpMeasure
		? gdpValues(gdp, data.countries)
		: countBy(data, 'country', { industries: nav.industries, sectors: nav.sectors });
	const total = gdpMeasure ? gdp.world.value : data.companies.length;
	const selected = nav.countries.length > 0 ? nav.countries : topIds(values, TOP_N);

	const sectorOf = dominantSector(data);
	const breakdown =
		nav.countries.length > 0
			? tilesOf(countBy(data, 'industry', { countries: nav.countries, sectors: nav.sectors }), (industry) => sectorFill(sectorOf[industry]))
			: null;
	const cross = nav.countries.length === 1 && nav.industries.length === 1 ? crossCell(data, nav.countries[0], nav.industries[0]) : null;

	const [from, to] = gdpYearRange(gdp);
	const missing = data.countries.filter((c) => !(c in values));
	const gdpNote = gdpMeasure
		? `GDP in current US$, the latest value per country (${from === to ? from : `${from} to ${to}`}). Source: World Bank, indicator NY.GDP.MKTP.CD, CC BY 4.0. ` +
			(missing.length > 0 ? `No GDP for: ${missing.join(', ')}.` : '')
		: null;

	return {
		measureLabel: gdpMeasure ? 'GDP' : 'companies',
		total,
		tiles: tilesOf(values, () => COUNTRY_FILL),
		shares: shareOfGlobal(values, selected, total),
		breakdown,
		breakdownTotal: data.companies.length,
		cross,
		gdpNote
	};
}
```

```ts
export interface IndustriesView {
	measureLabel: string;
	/** All companies: the global total the percentages refer to. */
	total: number;
	/** Every industry with companies, coloured by its sector. */
	tiles: TreemapItem[];
	/** The sectors present in `tiles`, with their colour and company count: the legend, and the drill-down links. */
	legend: { sector: string; fill: string; value: number }[];
	/** The selected industries (or the largest ten) against the global total. */
	shares: Shares;
	/** The countries of the selected industries, by company count; null when no industry is selected. */
	breakdown: TreemapItem[] | null;
	cross: CrossCell | null;
}

export function industriesView(data: CompanyData, nav: SizeNav): IndustriesView {
	const values = countBy(data, 'industry', { countries: nav.countries, sectors: nav.sectors });
	const total = data.companies.length;
	const sectorOf = dominantSector(data);
	const tiles = tilesOf(values, (industry) => sectorFill(sectorOf[industry]));
	const legendCounts = new Map<string, number>();
	for (const t of tiles) legendCounts.set(sectorOf[t.id], (legendCounts.get(sectorOf[t.id]) ?? 0) + t.value);
	const legend = [...legendCounts]
		.map(([sector, value]) => ({ sector, fill: sectorFill(sector), value }))
		.sort((a, b) => b.value - a.value || a.sector.localeCompare(b.sector));
	const selected = nav.industries.length > 0 ? nav.industries : topIds(values, TOP_N);
	const breakdown =
		nav.industries.length > 0 ? tilesOf(countBy(data, 'country', { industries: nav.industries, sectors: nav.sectors }), () => COUNTRY_FILL) : null;
	const cross = nav.countries.length === 1 && nav.industries.length === 1 ? crossCell(data, nav.countries[0], nav.industries[0]) : null;
	return { measureLabel: 'companies', total, tiles, legend, shares: shareOfGlobal(values, selected, total), breakdown, cross };
}
```

```ts
const data = companies as unknown as CompanyData;

// The page is cached for anonymous visitors; logged-in browsers skip the cache.
export const config = isrConfig(process.env.ISR_BYPASS_TOKEN);

export const load: PageServerLoad = ({ url, locals, setHeaders }) => {
	// Anonymous requests get an empty query, so what the cache stores never depends on who asked first.
	const params = pageParams(locals.interactive, url);
	const nav = parseSizeNav(params, { countries: data.countries, industries: data.industries, sectors: data.sectors }, '/industries');
	setHeaders({ 'cache-control': 'public, s-maxage=86400, stale-while-revalidate=86400' });
	return {
		nav,
		view: industriesView(data, nav),
		// the selector lists are only needed (and only shown) for logged-in visitors
		countries: locals.interactive ? data.countries : [],
		industries: locals.interactive ? data.industries : [],
		sectors: locals.interactive ? data.sectors : [],
		generated: data.generated
```
