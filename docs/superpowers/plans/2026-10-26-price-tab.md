**Goal:** Replace the "Coming soon" placeholder on the Stock Price tab of `/financials` with a summary of the last close and a line chart of the last year of daily closes.

**Architecture:** One FMP call (`historical-price-eod/full`, last year) through the shared `fmpGet`. `parsePrices`/`fetchPrices` live in `fmp-info.ts`. A pure `src/lib/prices.ts` computes the summary and formats numbers. `chart.ts` gains `priceTicks` (non-zero baseline) and `monthTicks`. A hand-written SVG `LineChart.svelte` draws the series. `loadFinancials` gets a `price` view backed by a 1 h cache; the `soon` state is removed because no tab uses it any more.

**Tech Stack:** SvelteKit 3 / Svelte 5, TypeScript (strict), Vitest, FMP stable API.

**Spec:** `docs/superpowers/specs/2026-10-26-price-tab-design.md` (builds on the financials-tabs and company-and-news specs).

```ts
const ISO_DATE = /^\d{4}-\d{2}-\d{2}$/;

export function parsePrices(body: unknown): PricePoint[] {
	throwIfErrorBody(body);
	if (!Array.isArray(body)) throw unexpected();
	const points: PricePoint[] = [];
	for (const entry of body) {
		if (!entry || typeof entry !== 'object') continue;
		const rec = entry as Record<string, unknown>;
		const date = str(rec.date);
		const close = num(rec.close);
		if (!date || !ISO_DATE.test(date) || close === null) continue;
		points.push({ date, close, volume: num(rec.volume) });
	}
	return points.sort((a, b) => (a.date < b.date ? -1 : a.date > b.date ? 1 : 0));
}
```

```ts
import type { PricePoint } from './server/fmp-info';

export interface PriceSummary {
	date: string;
	close: number;
	change: number | null;
	changePct: number | null;
	high: number;
	low: number;
	volume: number | null;
}

/** Summary of a series sorted oldest first; high and low are the highest and lowest close. */
export function summarise(points: readonly PricePoint[]): PriceSummary | null {
	const last = points.at(-1);
	if (!last) return null;
	const previous = points.length > 1 ? points[points.length - 2] : null;
	const closes = points.map((p) => p.close);
	const change = previous ? last.close - previous.close : null;
	return {
		date: last.date,
		close: last.close,
		change,
		changePct: change !== null && previous && previous.close !== 0 ? (change / previous.close) * 100 : null,
		high: Math.max(...closes),
		low: Math.min(...closes),
		volume: last.volume
	};
}

const iso = (ms: number) => new Date(ms).toISOString().slice(0, 10);

/** From one year before `today` to `today`, as UTC dates; 29 February maps to 28 February. */
export function oneYearWindow(today: Date): { from: string; to: string } {
	const y = today.getUTCFullYear();
	const m = today.getUTCMonth();
	const d = today.getUTCDate();
	let from = Date.UTC(y - 1, m, d);
	if (new Date(from).getUTCMonth() !== m) from = Date.UTC(y - 1, m + 1, 0);
	return { from: iso(from), to: iso(Date.UTC(y, m, d)) };
}

const TWO_DECIMALS = { minimumFractionDigits: 2, maximumFractionDigits: 2 } as const;

export function formatPrice(value: number): string {
	return value.toLocaleString('en', TWO_DECIMALS);
}

export function formatSigned(value: number): string {
	const text = Math.abs(value).toLocaleString('en', TWO_DECIMALS);
	return value > 0 && text !== '0.00' ? `+${text}` : value < 0 && text !== '0.00' ? `-${text}` : text;
}

export function formatVolume(value: number): string {
	return Math.round(value).toLocaleString('en');
}
```

```ts
/** Round tick values covering [min, max] with a little padding; unlike `niceTicks` the range need not include 0. */
export function priceTicks(
	min: number,
	max: number,
	count = 5
): { ticks: number[]; min: number; max: number } {
	const pad = max === min ? Math.abs(min) * 0.05 || 1 : (max - min) * 0.05;
	let lo = min - pad;
	const hi = max + pad;
	if (min >= 0) lo = Math.max(0, lo);

	const raw = (hi - lo) / (count - 1);
	const exp = Math.floor(Math.log10(raw));
	const fraction = raw / 10 ** exp;
	const step = (fraction <= 1 ? 1 : fraction <= 2 ? 2 : fraction <= 5 ? 5 : 10) * 10 ** exp;

	const niceMin = Math.floor(lo / step) * step;
	const niceMax = Math.ceil(hi / step) * step;
	const n = Math.round((niceMax - niceMin) / step);
	const ticks = Array.from({ length: n + 1 }, (_, i) => Number((niceMin + i * step).toFixed(10)));
	return { ticks, min: ticks[0], max: ticks[ticks.length - 1] };
}

const MONTHS = ['Jan', 'Feb', 'Mar', 'Apr', 'May', 'Jun', 'Jul', 'Aug', 'Sep', 'Oct', 'Nov', 'Dec'];
const MAX_MONTH_TICKS = 12;

/** The first point of each month, as x-axis labels; the year is added on the first label and on January. */
export function monthTicks(points: readonly { date: string }[]): { index: number; label: string }[] {
	let starts: { index: number; month: number; year: string }[] = [];
	points.forEach((p, index) => {
		const month = Number(p.date.slice(5, 7)) - 1;
		if (index === 0 || month !== Number(points[index - 1].date.slice(5, 7)) - 1) {
			starts.push({ index, month, year: p.date.slice(2, 4) });
		}
	});
	// a partial first month would put two labels almost on top of each other
	if (starts.length > 1 && starts[1].index - starts[0].index < points.length / 24) starts = starts.slice(1);
	const excess = starts.length - MAX_MONTH_TICKS;
	if (excess === 1) starts = starts.slice(1);
	else if (excess > 1) {
		const stride = Math.ceil(starts.length / MAX_MONTH_TICKS);
		starts = starts.filter((_, i) => i % stride === 0);
	}
	return starts.map((s, i) => ({
		index: s.index,
		label: i === 0 || s.month === 0 ? `${MONTHS[s.month]} '${s.year}` : MONTHS[s.month]
	}));
}
```

```svelte
<script lang="ts">
	import { PALETTE, monthTicks, priceTicks } from '#lib/chart';
	import { formatPrice } from '#lib/prices';
	import type { PricePoint } from '#lib/server/fmp-info';

	let { points }: { points: PricePoint[] } = $props();

	const W = 720;
	const H = 320;
	const M = { top: 14, right: 16, bottom: 30, left: 60 };
	const plotW = W - M.left - M.right;
	const plotH = H - M.top - M.bottom;
	/** Per-point hover targets get heavy beyond about a year of daily data. */
	const MAX_HOVER_POINTS = 260;

	const closes = $derived(points.map((p) => p.close));
	const scale = $derived(priceTicks(Math.min(...closes), Math.max(...closes)));
	const x = (i: number) => M.left + (points.length > 1 ? (plotW * i) / (points.length - 1) : plotW / 2);
	const y = (v: number) => M.top + (plotH * (scale.max - v)) / (scale.max - scale.min);
	const line = $derived(points.map((p, i) => `${x(i).toFixed(1)},${y(p.close).toFixed(1)}`).join(' '));
	const months = $derived(monthTicks(points));
	const summary = $derived(
		`Daily closing price from ${points[0].date} to ${points[points.length - 1].date}: ` +
			`from ${formatPrice(points[0].close)} to ${formatPrice(points[points.length - 1].close)}, ` +
			`low ${formatPrice(Math.min(...closes))}, high ${formatPrice(Math.max(...closes))}.`
	);
	const tickLabel = (v: number) => v.toLocaleString('en', { maximumFractionDigits: 2 });
</script>

<svg viewBox="0 0 {W} {H}" role="img" aria-label={summary}>
	{#each scale.ticks as tick (tick)}
		<line class="gridline" x1={M.left} x2={W - M.right} y1={y(tick)} y2={y(tick)} />
		<text class="tick" x={M.left - 6} y={y(tick) + 4} text-anchor="end">{tickLabel(tick)}</text>
	{/each}

	{#each months as m (m.index)}
		<line class="gridline month" x1={x(m.index)} x2={x(m.index)} y1={M.top} y2={H - M.bottom} />
		<text class="tick" x={x(m.index)} y={H - M.bottom + 18} text-anchor="middle">{m.label}</text>
	{/each}

	{#if points.length > 1}
		<polyline points={line} fill="none" stroke={PALETTE[0]} stroke-width="1.75" stroke-linejoin="round" />
	{:else}
		<circle cx={x(0)} cy={y(points[0].close)} r="4" fill={PALETTE[0]} />
	{/if}

	{#if points.length <= MAX_HOVER_POINTS}
		{#each points as p, i (p.date)}
			<circle class="hit" cx={x(i)} cy={y(p.close)} r="6">
				<title>{p.date}: {formatPrice(p.close)}</title>
			</circle>
		{/each}
	{/if}
</svg>
```
